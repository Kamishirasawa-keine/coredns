package blocklist

import (
	"os"
	"path/filepath"
	"testing"

	"github.com/miekg/dns"
)

func TestMultiFileAdGuardRules(t *testing.T) {
	var f filter
	dir := t.TempDir()
	for name, content := range map[string]string{
		"first":  "! Title\n||Ads.Example^\n@@||allow.ads.example^\n/^track[0-9]+\\.example$/\nexample.org##.ad\n",
		"second": "0.0.0.0 ADS.EXAMPLE.\n:: other.example\n||ignored.example^$client=10.0.0.0/8\n",
	} {
		path := filepath.Join(dir, name)
		if err := os.WriteFile(path, []byte(content), 0o600); err != nil {
			t.Fatal(err)
		}
		if _, err := f.load(path, false); err != nil {
			t.Fatal(err)
		}
	}
	if got := len(f.deny.domains); got != 2 {
		t.Fatalf("unique blocked domains = %d, want 2", got)
	}
	for name, want := range map[string]bool{
		"ads.example": true, "sub.ads.example": true, "notads.example": false,
		"allow.ads.example": false, "track42.example": true,
		"other.example": true, "ignored.example": false, "example.org": false,
	} {
		if got := f.blocked(name); got != want {
			t.Errorf("blocked(%s) = %v, want %v", name, got, want)
		}
	}
}

func TestAntiPCDN(t *testing.T) {
	path := filepath.Join(t.TempDir(), "adguard.txt")
	const content = `||*pcdn*.biliapi.net^$important
||mcdn.bilivideo.com^$important
@@||exception-pcdn.biliapi.net^
@@||important-pcdn.biliapi.net^$important
`
	if err := os.WriteFile(path, []byte(content), 0o600); err != nil {
		t.Fatal(err)
	}
	var f filter
	if n, err := f.load(path, false); err != nil || n != 4 {
		t.Fatalf("loaded %d rules, err=%v", n, err)
	}
	for name, want := range map[string]bool{
		"cache-pcdn-42.biliapi.net": true, "pcdn.biliapi.net": true,
		"mcdn.bilivideo.com": true, "exception-pcdn.biliapi.net": true,
		"important-pcdn.biliapi.net": false, "api.biliapi.net": false,
		"pcdn.biliapi.network": false, "pcdn.notbiliapi.net": false,
	} {
		if got := f.blocked(name); got != want {
			t.Errorf("blocked(%s) = %v, want %v", name, got, want)
		}
	}
}

func TestResponseModes(t *testing.T) {
	for _, tc := range []struct {
		mode string
		typ  uint16
		code int
		ip   string
	}{
		{"NXDOMAIN", dns.TypeA, dns.RcodeNameError, ""},
		{"NODATA", dns.TypeA, dns.RcodeSuccess, ""},
		{"REFUSED", dns.TypeA, dns.RcodeRefused, ""},
		{"ZERO", dns.TypeA, dns.RcodeSuccess, "0.0.0.0"},
		{"ZERO", dns.TypeAAAA, dns.RcodeSuccess, "::"},
		{"ZERO", dns.TypeTXT, dns.RcodeSuccess, ""},
	} {
		r, err := parseResponse(tc.mode)
		if err != nil {
			t.Fatal(err)
		}
		q := new(dns.Msg).SetQuestion("blocked.example.", tc.typ)
		m := r.reply(q)
		var ip string
		if len(m.Answer) > 0 {
			switch record := m.Answer[0].(type) {
			case *dns.A:
				ip = record.A.String()
			case *dns.AAAA:
				ip = record.AAAA.String()
			default:
				t.Fatalf("unexpected answer %v", record)
			}
		}
		if m.Rcode != tc.code || ip != tc.ip {
			t.Errorf("%s / %d: rcode=%d ip=%q, want %d %q", tc.mode, tc.typ, m.Rcode, ip, tc.code, tc.ip)
		}
	}
}
